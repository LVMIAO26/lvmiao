核心代码

<!-- 1. 引入 Lucide JS --> <script src="./cdn/lucide/lucide.min.js"></script> <!-- 2. 使用 data-lucide 声明图标 --> <i data-lucide="heart"></i> <i data-lucide="heart" width="32" height="32" style="color:#6c5ce7"></i> <!-- 3. 调用 createIcons() 将 <i> 替换为内联 <svg> --> <script>lucide.createIcons();</script>
