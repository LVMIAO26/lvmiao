一、条件语句 if...else

根据条件真假选择不同分支执行：输入分数 0~100，自动评定等级。

if (lm_s >= 90)      { lm_grade = "优秀"; }
else if (lm_s >= 80) { lm_grade = "良好"; }
else if (lm_s >= 60) { lm_grade = "及格"; }
else                 { lm_grade = "不及格"; }

二、条件语句 switch

多分支精确匹配，比连续 if...else 更清晰：输入 1~7 输出对应星期。

switch (lm_n) {
    case 1: lm_day = "星期一"; break;
    case 2: lm_day = "星期二"; break;
    ...
    default: lm_day = "输入无效";
}

三、循环语句 for

已知循环次数时使用：累加 1~100，求 1+2+3+…+100 的和。

var lm_sum = 0;
for (var lm_i = 1; lm_i <= 100; lm_i++) {
    lm_sum += lm_i;      // 循环累加 100 次
}

四、循环语句 while

先判断后执行，条件为真就继续：用 while 计算阶乘 6! = 720。

var lm_f = 1, lm_n = 6;
while (lm_n > 1) {
    lm_f *= lm_n;        // 6*5*4*3*2*1
    lm_n--;
}

五、循环语句 do...while

先执行一次再判断，循环体至少执行一次：输出 1~10。

var lm_i = 1, lm_out = "";
do {
    lm_out += lm_i + " ";
    lm_i++;
} while (lm_i <= 10);

六、break 与 continue

break 立即跳出整个循环；continue 跳过本次循环进入下一次。

if (lm_i % 7 === 0) { break; }    // 找到即退出
if (lm_i % 2 === 0) { continue; } // 跳过偶数

七、嵌套循环 · 九九乘法表

外层循环控制行（被乘数 1~9），内层循环控制列（乘数 1~行数），页面加载时自动渲染。

for (var lm_i = 1; lm_i <= 9; lm_i++) {        // 外层：行
    for (var lm_j = 1; lm_j <= lm_i; lm_j++) {  // 内层：列
        lm_html += lm_j + "×" + lm_i + "=" + (lm_i * lm_j);
    }
}
